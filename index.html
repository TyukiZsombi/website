    <!-- Three.js Shader Text Logic & Interactions -->

    <script>

        document.addEventListener('DOMContentLoaded', () => {

            // --- EGYEDI KURZOR LOGIKA ---

            const cursor = document.getElementById('custom-cursor');

            const hoverables = document.querySelectorAll('.hoverable, a, button, input, textarea');


            document.addEventListener('mousemove', (e) => {

                cursor.style.left = e.clientX + 'px';

                cursor.style.top = e.clientY + 'px';

            });


            hoverables.forEach(el => {

                el.addEventListener('mouseenter', () => document.body.classList.add('hover-active'));

                el.addEventListener('mouseleave', () => document.body.classList.remove('hover-active'));

            });


            // --- GSAP ANIMÁCIÓK ---

            gsap.registerPlugin(ScrollTrigger);


            const animatedElements = document.querySelectorAll('.animate-on-scroll');

            animatedElements.forEach(el => {

                gsap.fromTo(el, 

                    { y: 50, opacity: 0 },

                    { 

                        y: 0, 

                        opacity: 1, 

                        duration: 1, 

                        ease: "power3.out",

                        scrollTrigger: {

                            trigger: el,

                            start: "top 85%",

                            toggleActions: "play none none reverse"

                        }

                    }

                );

            });


            // --- THREE.JS SHADER ALAPÚ ANIMÁLT SZÖVEG ---

            const container = document.getElementById('shader-container');

            

            // Színtér, Kamera, Renderer

            const scene = new THREE.Scene();

            // Orthographic camera hogy pixel-perfect legyen a vászonhoz

            const camera = new THREE.OrthographicCamera( -1, 1, 1, -1, 0.1, 10 );

            camera.position.z = 1;


            const renderer = new THREE.WebGLRenderer({ alpha: true, antialias: true });

            renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));

            renderer.setSize(container.clientWidth, container.clientHeight);

            container.appendChild(renderer.domElement);


            // 2D Canvas a szöveg generálására (ez lesz a textúra)

            const textCanvas = document.createElement('canvas');

            const textCtx = textCanvas.getContext('2d');

            let textTexture = new THREE.CanvasTexture(textCanvas);

            textTexture.minFilter = THREE.LinearFilter;

            textTexture.magFilter = THREE.LinearFilter;


            function updateTextCanvas() {

                const w = container.clientWidth;

                const h = container.clientHeight;

                const dpr = Math.min(window.devicePixelRatio, 2);

                

                textCanvas.width = w * dpr;

                textCanvas.height = h * dpr;

                

                textCtx.scale(dpr, dpr);

                textCtx.clearRect(0, 0, w, h);


                // Dinamikus, sosem kilógó betűméret számítás

                const text = "INVIOS";

                let fontSize = 300;

                textCtx.font = `900 ${fontSize}px Syne, sans-serif`;

                let textWidth = textCtx.measureText(text).width;

                

                // Csökkentjük a méretet, amíg be nem fér (max 90% szélesség)

                while(textWidth > w * 0.9 && fontSize > 20) {

                    fontSize -= 5;

                    textCtx.font = `900 ${fontSize}px Syne, sans-serif`;

                    textWidth = textCtx.measureText(text).width;

                }


                // Szöveg rajzolása (Középre, a viewport felső részébe tolva picit)

                textCtx.fillStyle = '#2A1B14'; // Espresso szín, amit majd a shader torzít és színez

                textCtx.textAlign = 'center';

                textCtx.textBaseline = 'middle';

                textCtx.fillText(text, w / 2, h / 2 - 50);


                textTexture.needsUpdate = true;

            }


            // Betűtípus betöltés megvárása, utána rajzolás

            document.fonts.ready.then(() => {

                updateTextCanvas();

            });


            // Shader Material a torzításhoz

            const vertexShader = `

                varying vec2 vUv;

                void main() {

                    vUv = uv;

                    gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);

                }

            `;


            const fragmentShader = `

                varying vec2 vUv;

                uniform sampler2D u_texture;

                uniform float u_time;

                uniform vec2 u_mouse;

                uniform vec2 u_resolution;


                void main() {

                    vec2 uv = vUv;


                    // Groovy / Hullámos torzítás a szövegen idő alapján

                    float waveX = sin(uv.y * 15.0 + u_time * 2.0) * 0.015;

                    float waveY = cos(uv.x * 10.0 + u_time * 1.5) * 0.015;

                    

                    uv.x += waveX;

                    uv.y += waveY;


                    // Egér interakció (taszítja a pixeleket)

                    float dist = distance(uv, u_mouse);

                    float effectRadius = 0.2;

                    if(dist < effectRadius) {

                        float force = (effectRadius - dist) / effectRadius;

                        vec2 dir = normalize(uv - u_mouse);

                        uv += dir * force * 0.05; // Taszítás erőssége

                    }


                    // Textúra mintavételezés a torzított koordinátákkal

                    vec4 color = texture2D(u_texture, uv);


                    // Ha a pixel nem átlátszó (tehát a szöveg része), alkalmazzunk egy finom meleg gradienst rajta

                    if(color.a > 0.1) {

                        // Korall és sárga színátmenet a szövegen belül (mix a base espresso színnel)

                        vec3 gradientColor = mix(vec3(1.0, 0.49, 0.31), vec3(1.0, 0.84, 0.0), uv.x + waveX); // Coral to Yellow

                        

                        // Kicsit rátesszük a textúrára

                        color.rgb = mix(color.rgb, gradientColor, 0.4 + sin(u_time)*0.1); 

                    }


                    gl_FragColor = color;

                }

            `;


            const uniforms = {

                u_texture: { value: textTexture },

                u_time: { value: 0.0 },

                u_mouse: { value: new THREE.Vector2(-1, -1) }, // Kezdetben képernyőn kívül

                u_resolution: { value: new THREE.Vector2(container.clientWidth, container.clientHeight) }

            };


            const material = new THREE.ShaderMaterial({

                vertexShader: vertexShader,

                fragmentShader: fragmentShader,

                uniforms: uniforms,

                transparent: true

            });


            // Plane ami kitölti a kamerát

            const geometry = new THREE.PlaneGeometry(2, 2);

            const plane = new THREE.Mesh(geometry, material);

            scene.add(plane);


            // Egérkövetés a shaderhez

            window.addEventListener('mousemove', (e) => {

                // Konvertálás Normalized Device Coordinates-ba (0.0 to 1.0 a shadernek)

                uniforms.u_mouse.value.x = e.clientX / window.innerWidth;

                uniforms.u_mouse.value.y = 1.0 - (e.clientY / window.innerHeight);

            });


            // Érintőképernyő támogatás

            window.addEventListener('touchmove', (e) => {

                if(e.touches.length > 0) {

                    uniforms.u_mouse.value.x = e.touches[0].clientX / window.innerWidth;

                    uniforms.u_mouse.value.y = 1.0 - (e.touches[0].clientY / window.innerHeight);

                }

            });


            // Ablak átméretezés kezelése

            window.addEventListener('resize', () => {

                const w = container.clientWidth;

                const h = container.clientHeight;

                

                renderer.setSize(w, h);

                uniforms.u_resolution.value.set(w, h);

                updateTextCanvas();

            });


            // Animációs ciklus

            const clock = new THREE.Clock();

            function animate() {

                requestAnimationFrame(animate);

                uniforms.u_time.value = clock.getElapsedTime();

                renderer.render(scene, camera);

            }

            animate();

        });
